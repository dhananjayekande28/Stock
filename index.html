async function fetchData(symbol){ const url="https://proxy.cors.dev/https://query1.finance.yahoo.com/v8/finance/chart/"+ encodeURIComponent(symbol)+"?range=2y&interval=1d"; const response=await fetch(url); if(!response.ok) throw Error("डेटा उपलब्ध नहीं"); const json=await response.json(); const result=json.chart.result?.[0]; if(!result) throw Error("भाव उपलब्ध नहीं"); const meta=result.meta; const raw=result.indicators.quote[0].close; // केवल वास्तविक उपलब्ध भाव रखें const valid=raw .map((v,i)=>({price:v,time:result.timestamp?.[i]})) .filter(x=>Number.isFinite(x.price)); if(valid.length<30) throw Error("पर्याप्त डेटा नहीं"); // एक ही डेटा स्रोत के लगातार दो ट्रेडिंग सत्र const latest=valid[valid.length-1]; const previous=valid[valid.length-2]; const price=latest.price; const prev=previous.price; if(!Number.isFinite(price)||!Number.isFinite(prev)||prev===0) throw Error("भाव की पुष्टि नहीं हो सकी"); const closes=valid.map(x=>x.price); const daily=indicators(closes); const weekly=indicators(aggregate(closes,5)); const monthly=indicators(aggregate(closes,21)); daily.price=price; weekly.price=price; monthly.price=price; return { price, prev, change:price-prev, pct:((price-prev)/prev)*100, daily, weekly, monthly, time:latest.time ?new Date(latest.time*1000).toLocaleString("en-IN") :"उपलब्ध नहीं" }; }
